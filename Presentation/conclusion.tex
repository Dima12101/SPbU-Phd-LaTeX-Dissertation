\section{Итоги работы}

\begin{frame}
	\frametitle{Ограничения и дальнейшая работа}
	\small
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.48\textwidth}
			\textbf{Ограничения}
			\begin{itemize}
				\item свойства VCG относятся к отдельному раунду при фиксированных режимах;
				\item платежи и выбор победителей могут быть дороги на крупных масштабах;
				\item качество политики зависит от сценариев обучения;
				\item кластеризация полезна только при достаточном масштабе и локальности графа.
			\end{itemize}
		\end{column}
		\begin{column}{0.48\textwidth}
			\textbf{Дальнейшие направления}
			\begin{itemize}
				\item выбор оркестратора распределения;
				\item приближённые VCG-платежи;
				\item более сложные ограничения и нелинейные стоимости;
				\item перенос политики между конфигурациями;
				\item проверка в более детализированных робототехнических симуляторах.
			\end{itemize}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}
	\frametitle{Основные результаты}
	\small
	\begin{enumerate}
		\item Завершён текст диссертации и согласована структура исследования.
		\item Разработаны математическая модель, аукционный слой, MARL-формализация и гибридный метод AMPLE-AMR.
		\item Разработана масштабируемая версия C-AMPLE-AMR на основе кластеризации графа инфраструктуры.
		\item Реализован программный прототип и набор сценариев роботизированного склада.
		\item Проведена экспериментальная проверка эффективности, устойчивости и масштабируемости метода.
		\item Сформулированы ограничения метода и направления дальнейшего развития.
	\end{enumerate}
\end{frame}

\begin{frame}
	\frametitle{Положения, выносимые на защиту}
	\scriptsize
	\begin{enumerate}
		\item \textbf{Математическая модель AMR-системы} с поддержкой периферийных вычислений, описывающая распределение вычислительных задач между гетерогенными периферийными узлами с учётом характеристик задач, приоритетов и директивных сроков выполнения, сетевых задержек, очередей, вычислительных ресурсов и мобильности роботов.
		\item \textbf{Формализация задачи адаптивного распределения вычислительных задач} автономных мобильных роботов как задачи внутренней координации общего дефицитного вычислительного ресурса, включающая механизм аукционного распределения Викри–Кларка–Гровса для выбора исполнителей задач и вычисления внутренних теневых цен создаваемых внешних эффектов.
		\item \textbf{Гибридный метод AMPLE-AMR} адаптивного распределения вычислительных задач автономных мобильных роботов, объединяющий аукционный механизм распределения задач и кооперативную многоагентную политику управления операционными режимами периферийных узлов, формализованную в виде децентрализованного частично наблюдаемого марковского процесса принятия решений.
		\item \textbf{Масштабируемый метод C-AMPLE-AMR}, развивающий AMPLE-AMR посредством разбиения графа периферийной инфраструктуры на кластеры и выполнения локальных аукционных процедур распределения задач внутри кластеров, что обеспечивает снижение вычислительных и коммуникационных затрат при увеличении числа роботов и периферийных узлов.
	\end{enumerate}
\end{frame}
