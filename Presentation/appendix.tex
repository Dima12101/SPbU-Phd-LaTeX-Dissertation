\begin{frame}
	\frametitle{Приложение: поведение обученной политики}
	\small
	\begin{center}
		\maybeincludegraphics[width=\linewidth,height=0.58\textheight,keepaspectratio]{Dissertation/images/results/ch8_operation_mode_distribution}
	\end{center}
	\begin{itemize}
		\item В стабильной нагрузке доминирует \texttt{normal} ($\approx 70\%$), а \texttt{aggressive} и \texttt{critical} используются как усилители.
		\item При пиковой нагрузке доля \texttt{aggressive} возрастает примерно до $30\%$.
		\item В разнородной инфраструктуре почти весь выбор смещается в \texttt{reserve} ($\approx 98\%$).
		\item В сценарии clustered/global для Warehouse-L почти весь выбор переходит в \texttt{critical}.
		\item Это подтверждает адаптацию к состоянию среды, а не случайное переключение.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Приложение: общественное благосостояние}
	\small
	\begin{center}
		\maybeincludegraphics[width=\linewidth,height=0.48\textheight,keepaspectratio]{Dissertation/images/results/ch8_welfare_by_scenario}
	\end{center}
	\begin{itemize}
		\item Fixed-Auction превосходит Fixed-Heuristic по $SW$ во всех сценариях и во всех сопоставлениях H1.
		\item В стабильной нагрузке: $634.4 \rightarrow 1168.6 \rightarrow 1235.6$ для Fixed-Heuristic, Fixed-Auction и AMPLE-AMR.
		\item При пиковой нагрузке: $957.8 \rightarrow 2315.9 \rightarrow 2516.7$.
		\item В разнородной инфраструктуре лучшим по $SW$ оказывается Fixed-Auction ($1248.2$), а не AMPLE-AMR ($1183.1$).
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Приложение: масштабируемость}
	\small
	\begin{center}
		\maybeincludegraphics[width=\linewidth,height=0.48\textheight,keepaspectratio]{Dissertation/images/results/ch8_overhead_scalability}
	\end{center}
	\begin{itemize}
		\item На Warehouse-S/M/L глобальный AMPLE-AMR остаётся практически приемлемым, а кластеризация чаще невыгодна.
		\item На Warehouse-XL накладные расходы AMPLE-AMR возрастают до $264.1$~мс и становятся узким местом управляющего слота.
		\item C-AMPLE-AMR снижает overhead до $51.3$~мс, то есть примерно на $80.5\%$.
		\item Потеря благосостояния на Warehouse-XL составляет $41.9$, поэтому кластеризация полезна только на экстремальном масштабе.
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Приложение: качество обслуживания: задержка и отказы}
	\small
	\begin{itemize}
		\item В стабильной нагрузке baseline Fixed-Heuristic остаётся сильным по throughput и хвосту задержки.
		\item При пиковой нагрузке P95 уменьшается с $78.6$ до $66.4$~мс при переходе от Fixed-Heuristic к AMPLE-AMR.
		\item В сценариях деградации связи и отказов P95 снижается соответственно с $70.1$ до $62.8$~мс и с $72.5$ до $58.7$~мс.
		\item Улучшение $SW$ и хвоста задержки у аукционных методов сопровождается меньшим completion rate и большим drop rate.
		\item Доля нарушений сроков почти во всех сценариях близка к нулю из-за фильтрации недопустимых назначений.
	\end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Приложение: сравнение глобального и кластерного вариантов}
    \small
    \begin{center}
    	\maybeincludegraphics[width=\linewidth,height=0.48\textheight,keepaspectratio]{Dissertation/images/results/ch8_clustered_tradeoff}
    \end{center}
   	\begin{itemize}
	    \item На Warehouse-L глобальный AMPLE-AMR предпочтительнее: $3.29$ против $7.63$~мс при потере welfare $9.0$ у кластерного варианта.
	    \item На Warehouse-XL кластерный вариант снижает overhead с $264.1$ до $51.3$~мс.
	    \item Цена ускорения на Warehouse-XL составляет потерю welfare $41.9$.
	    \item Переход к C-AMPLE-AMR должен зависеть от доли управляющего слота, занятой глобальным распределением.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Приложение: чувствительность к режимам}
    \small
    \begin{itemize}
        \item Для AMPLE-AMR профиль \texttt{aggressive\_bias} даёт наибольшее общественное благосостояние: $977.0$ против $948.7$ в default.
        \item Для C-AMPLE-AMR картина похожа: $976.8$ в \texttt{aggressive\_bias} против $966.9$ в default.
        \item Консервативный профиль снижает $SW$ у глобального и кластерного гибридных методов.
        \item Во всех проверенных профилях гибридные методы остаются выше фиксированных базовых методов по общественному благосостоянию.
    \end{itemize}
\end{frame}
