% Локальная команда для безопасного подключения графиков из диссертации.
% Если файл отсутствует в текущей сборке, вместо ошибки будет показана рамка-заглушка.
\newcommand{\maybeincludegraphics}[2][]{%
    \IfFileExists{#2.pdf}{\includegraphics[#1]{#2}}{%
    \IfFileExists{#2.png}{\includegraphics[#1]{#2}}{%
    \IfFileExists{#2.jpg}{\includegraphics[#1]{#2}}{%
    \IfFileExists{#2.jpeg}{\includegraphics[#1]{#2}}{%
    \fbox{\begin{minipage}[c][0.35\textheight][c]{0.86\linewidth}\centering\scriptsize Рисунок не найден:\\[-1mm]\texttt{\detokenize{#2}}\end{minipage}}%
    }}}}%
}
