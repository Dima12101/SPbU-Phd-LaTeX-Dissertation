\chapter{Метрики эффективности системы}\label{app:А}

\begin{table}[h]
	\centering
	\caption{Метрики оценки эффективности системы}
	\label{tab:evaluation_metrics}
	\small
	\begin{tabular}{p{2.5cm}p{4cm}p{7.5cm}}
		\toprule
		\multicolumn{2}{l}{\textbf{Метрика}} & \textbf{Формула и интерпретация} \\
		\midrule
		Процент принятых задач & 
		AcceptanceRate & 
		$\text{AcceptanceRate} = \frac{N_{\text{accepted}}}{N_{\text{total}}} \cdot 100\%$ \\
		& & \textit{Доля задач, принятых системой; важен при высокой нагрузке;} \\
		& & \textit{Выше -- лучше (цель: $>90\%$)} \\
		\midrule
		Выполнение дедлайна & 
		DeadlineRate & 
		$\text{DeadlineRate} = \frac{N_{\text{accepted}}}{N_{\text{total}}} \cdot 100\%$ \\
		& & \textit{Доля завершённых задач, уложившихся
			в дедлайн;} \\
		& & \textit{Выше -- лучше (цель: $>90\%$)} \\
		\midrule
		Средняя задержка & 
		AvgLatency & 
		$\text{AvgLatency} = \frac{1}{T} \sum_{t=1}^{T} \text{Latency}(t)$ \\
		& & \textit{Среднее время от прихода задачи до завершения;} \\
		& & \textit{Меньше -- лучше (цель: $<150$ мс)} \\
		\midrule
		Джини платежей & 
		Gini$(p)$ & 
		$\text{Gini}(p) = \frac{\sum_{i=1}^{m} \sum_{j=1}^{m} |p_i - p_j|}{2 m^2 \bar{p}}$ \\
		& & 
		\textit{Справедливо: $<0.3$} \\
		\midrule
		Индекс справедливости & 
		FairnessIndex & 
		$\text{FairnessIndex} = \frac{(\sum_{i=1}^{n} x_i)^2}{n \sum_{i=1}^{n} x_i^2}$ \\
		& & \textit{Индекс Джини для распределения нагрузки или ресурсов между узлами;} \\
		& & \textit{Справедливо: $>0.85$} \\
		\midrule
		Дисбаланс загрузки узлов $i$ & 
		LoadImbalance$_i$ & 
		$\text{LoadImbalance}_i = \frac{\sum_{t=1}^{T} \text{Load}_i(t)}{T \cdot \text{Capacity}_i}$ \\
		& & \textit{Средняя дисбаланс загрузки вычислительных ресурсов узла $i$;} \\
		& & \textit{Оптимально: $0.8$--$0.9$} \\
		\bottomrule
	\end{tabular}
\end{table}
