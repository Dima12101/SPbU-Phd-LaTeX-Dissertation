\chapter{Метрики качества}\label{app:Б}

\textbf{Общественное благосостояние:}
\[
SW(t)=
\sum_{\tau}
\sum_j
x_{\tau j}(t)
\left[
U_{i(\tau)}(\tau,n_j,t)
-
C_j(\tau,r_{i(\tau)},t)
\right].
\]

\textbf{Доля обслуженных задач:}
\[
CompletionRate =
\frac{N_{completed}}{N_{generated}}.
\]

\textbf{Доля отклонённых задач:}
\[
DropRate =
\frac{N_{dropped}}{N_{generated}}.
\]

\textbf{Доля нарушений сроков:}
\[
ViolationRate =
\frac{N_{violated}}{N_{completed}}.
\]

\textbf{Средняя задержка обработки:}
\[
AvgLatency =
\frac{1}{N_{completed}}
\sum_{\tau\in Completed}
T_{\tau}.
\]

\textbf{95-й квантиль задержки:}
\[
P95Latency =
\mathrm{quantile}_{0.95}
\left(
\{T_{\tau}: \tau\in Completed\}
\right).
\]

\textbf{Средняя загрузка узлов:}
\[
AvgUtilization =
\frac{1}{k}
\sum_{j=1}^{k}
u_j.
\]

\textbf{Дисбаланс загрузки:}
\[
LoadImbalance =
\frac{1}{k}
\sum_{j=1}^{k}
(u_j-\bar{u})^2.
\]

\textbf{Накладные расходы распределения:}
\[
AllocationOverhead =
T^{auction}
+
T^{comm}
+
T^{policy}.
\]

\textbf{Доля управляющего слота, занятая распределением:}
\[
SlotShare =
\frac{
	T^{auction}+T^{comm}+T^{policy}
}{
	\Delta^{slot}
}.
\]

Для C-AMPLE-AMR дополнительно оцениваются:
\begin{enumerate}
	\item число кластеров;
	\item средний размер кластера;
	\item время локальных распределений;
	\item оценка потери общественного благосостояния относительно глобального
	AMPLE-AMR;
	\item величина межкластерного среза графа.
\end{enumerate}
